from datetime import datetime
from typing import List, Optional
from fastapi import FastAPI, HTTPException, Depends
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from sqlalchemy import create_engine, Column, Integer, String, DateTime
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, Session

# Configuración de base de datos SQLite
DATABASE_URL = "sqlite:///./pendientes.db"
engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# Modelo de Base de Datos
class PendienteDB(Base):
    __tablename__ = "pendientes"

    id = Column(Integer, primary_key=True, index=True)
    descripcion = Column(String, nullable=False)
    importancia = Column(String, nullable=False) # Alta, Media, Baja
    estado = Column(String, default="pendiente") # pendiente, completado
    fecha_creacion = Column(DateTime, default=datetime.now)
    fecha_atencion = Column(DateTime, nullable=True)

Base.metadata.create_all(bind=engine)

# Esquemas de Pydantic
class PendienteCreate(BaseModel):
    descripcion: str
    importancia: str

class PendienteResponse(BaseModel):
    id: int
    descripcion: str
    importancia: str
    estado: str
    fecha_creacion: datetime
    fecha_atencion: Optional[datetime] = None

    class Config:
        from_attributes = True

# Inicialización de la aplicación FastAPI
app = FastAPI(title="Gestor de Pendientes")

# Permitir solicitudes CORS para desarrollo frontend
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Dependencia de sesión de BD
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# Rutas API
@app.get("/pendientes", response_model=List[PendienteResponse])
def listar_pendientes(db: Session = Depends(get_db)):
    return db.query(PendienteDB).order_by(PendienteDB.fecha_creacion.desc()).all()

@app.post("/pendientes", response_model=PendienteResponse)
def crear_pendiente(item: PendienteCreate, db: Session = Depends(get_db)):
    if item.importancia not in ["Alta", "Media", "Baja"]:
        raise HTTPException(status_code=400, detail="Importancia no válida")
    
    nuevo = PendienteDB(
        descripcion=item.descripcion,
        importancia=item.importancia,
        fecha_creacion=datetime.now()
    )
    db.add(nuevo)
    db.commit()
    db.refresh(nuevo)
    return nuevo

@app.patch("/pendientes/{item_id}/estado", response_model=PendienteResponse)
def cambiar_estado(item_id: int, db: Session = Depends(get_db)):
    item = db.query(PendienteDB).filter(PendienteDB.id == item_id).first()
    if not item:
        raise HTTPException(status_code=404, detail="Pendiente no encontrado")

    if item.estado == "pendiente":
        item.estado = "completado"
        item.fecha_atencion = datetime.now()
    else:
        item.estado = "pendiente"
        item.fecha_atencion = None

    db.commit()
    db.refresh(item)
    return item

@app.delete("/pendientes/{item_id}")
def eliminar_pendiente(item_id: int, db: Session = Depends(get_db)):
    item = db.query(PendienteDB).filter(PendienteDB.id == item_id).first()
    if not item:
        raise HTTPException(status_code=404, detail="Pendiente no encontrado")
    db.delete(item)
    db.commit()
    return {"message": "Eliminado exitosamente"}
